"screw_standoff_diameter" = 4 ' diameter of the boss under each PCB corner that fills the pedestal_depth gap, so the screw clamps the board flat instead of bending it into the gap

"screw_hole_runout" = 1 ' how far the tap-drill hole goes beyond the engaged thread (the drill point and the tap chamfer)

"screw_floor_skin" = 0.8 ' minimum metal left under the bottom of the blind threaded hole
